import { getAuth, signInWithEmailAndPassword, signInAnonymously, signOut, onAuthStateChanged } from "https://www.gstatic.com/firebasejs/12.14.0/firebase-auth.js"; const auth = getAuth(app); /* UI */ const roleTag = document.getElementById("roleTag"); /* ADMIN IDS */ const ADMIN_UIDS = [ "j4KFCYgFz0UR6n2NF2ZdLkwUfkm1" ]; /* AUTO GUEST LOGIN (IMPORTANT FIX) */ signInAnonymously(auth); // 👈 this makes guest automatic on load /* LOGIN BUTTONS (still optional upgrade path) */ document.getElementById("loginBtn").onclick = async () => { await signInWithEmailAndPassword( auth, document.getElementById("email").value, document.getElementById("pass").value ); }; document.getElementById("logoutBtn").onclick = () => signOut(auth); /* AUTH STATE */ onAuthStateChanged(auth, u => { if(!u){ roleTag.textContent = "[connecting...]"; return; } let role; if (u.isAnonymous) { role = "guest"; } else { role = ADMIN_UIDS.includes(u.uid) ? "admin" : "operator"; } roleTag.textContent = `[${role}] ${u.email || "guest"}`; window.currentUser = u; window.currentRole = role; setupPresence(); listen(); });
